function getSheet(name) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); return ss.getSheetByName(name) || ss.insertSheet(name); } // Respuesta con headers CORS explícitos function respond(data) { const output = ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); return output; } function doGet(e) { try { const action = (e.parameter && e.parameter.action) ? e.parameter.action : 'load'; if (action === 'ping') return respond({ ok: true, msg: 'OK' }); if (action === 'load') return respond({ ok: true, data: loadAll() }); } } // ── SAVE ──────────────────────────────────────────────────────────────────── function saveAll(s) { if (s.transactions) saveRows('transactions', ['id','date','desc','type','amount','account','presCat','note'], (s.transactions||[]).map(t=>[t.id,t.date,t.desc,t.type,t.amount,t.account,t.presCat??'',t.note??''])); if (s.cajitas) saveRows('cajitas', ['name','desc','saldo','ganancias','retirada','isTDC','combinadoCon'],(s.cajitas||[]).map(c=>[c.name,c.desc,c.saldo,c.ganancias,c.retirada,c.isTDC?'true':'false',c.combinadoCon??''])); if (s.deudas != null) saveRows('deudas', ['quien','concepto','monto','fecha','vence','nota','status'], (s.deudas||[]).map(d=>[d.quien,d.concepto,d.monto,d.fecha,d.vence??'',d.nota??'',d.status])); if (s.ingresoEsperado) saveJson('ingresoEsperado', s.ingresoEsperado); if (s.saldoInicialMes) saveJson('saldoInicialMes', s.saldoInicialMes); getSheet('log').appendRow([new Date().toISOString(), 'SAVE']); } function saveRows(tab, headers, rows) { const sh = getSheet(tab); } // ── LOAD ──────────────────────────────────────────────────────────────────── function loadAll() { return { transactions: loadRows('transactions', r=>({id:r[0],date:r[1],desc:r[2],type:r[3],amount:+r[4]||0,account:r[5],presCat:r[6]!==''?+r[6]:null,note:r[7]||''})), presupuesto: loadPresupuesto(), cuentas: loadRows('cuentas', r=>({id:r[0],name:r[1],saldo:+r[2]||0,type:r[3],class:r[4],corte:r[5]!==''?+r[5]:undefined,pago:r[6]!==''?+r[6]:undefined})), cajitas: loadRows('cajitas', r=>({name:r[0],desc:r[1],saldo:+r[2]||0,ganancias:+r[3]||0,retirada:+r[4]||0,isTDC:r[5]==='true',combinadoCon:r[6]||null})), deudas: loadRows('deudas', r=>({quien:r[0],concepto:r[1],monto:+r[2]||0,fecha:r[3],vence:r[4]||null,nota:r[5]||'',status:r[6]})), ingresoEsperado: loadJson('ingresoEsperado'), saldoInicialMes: loadJson('saldoInicialMes'), }; } function loadRows(tab, mapper) { try { const data = getSheet(tab).getDataRange().getValues(); return data.length > 1 ? data.slice(1).map(mapper) : []; } catch(e) { return []; } } function loadPresupuesto() { try { const rows = getSheet('presupuesto').getDataRange().getValues(); if (rows.length <= 1) return null; const p = { needs:[], savings:[], personal:[] }; rows.slice(1).forEach(r => { if(p[r[0]]) p[r[0]].push({id:+r[1],name:r[2],bud:+r[3]||0,real:+r[4]||0}); }); return p; } catch(e) { return null; } } function loadJson(tab) { try { const val = getSheet(tab).getRange('A1').getValue(); return val ? JSON.parse(val) : null; } catch(e) { return null; } }
